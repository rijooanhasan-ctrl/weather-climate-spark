WITH base AS (
        SELECT *
        FROM weather_clean
        WHERE 1 = 1
    ),
    monthly_agg AS (
        SELECT
            location,
            year,
            month,
            ROUND(AVG(temperature), 2) AS avg_temp,
            ROW_NUMBER() OVER (PARTITION BY location ORDER BY year, month) AS month_idx,
            AVG(AVG(temperature)) OVER (
                PARTITION BY location
                ORDER BY year, month
                ROWS BETWEEN 11 PRECEDING AND CURRENT ROW
            ) AS rolling_12m_raw
        FROM base
        GROUP BY location, year, month
    )
    SELECT
        location,
        year,
        month,
        avg_temp,
        CASE
            WHEN month_idx >= 12 THEN ROUND(rolling_12m_raw, 2)
            ELSE NULL
        END AS rolling_12m_avg
    FROM monthly_agg
    ORDER BY location, year, month

WITH base AS (
        SELECT *
        FROM weather_clean
        WHERE 1 = 1
    ),
    loc_summary AS (
        SELECT
            location,
            ROUND(AVG(temperature), 2) AS avg_temp,
            MAX(temp_max) AS max_val,
            MIN(temp_min) AS min_val
        FROM base
        GROUP BY location
    ),
    extremes_dates AS (
        SELECT
            b.location,
            FIRST(CASE WHEN b.temp_max = s.max_val THEN CAST(b.date AS STRING) END, true) AS record_max_date,
            FIRST(CASE WHEN b.temp_min = s.min_val THEN CAST(b.date AS STRING) END, true) AS record_min_date
        FROM base b
        JOIN loc_summary s ON b.location = s.location
        GROUP BY b.location
    )
    SELECT
        s.location,
        s.avg_temp,
        ROUND(s.max_val, 2) AS record_max,
        d.record_max_date,
        ROUND(s.min_val, 2) AS record_min,
        d.record_min_date,
        RANK() OVER (ORDER BY s.avg_temp DESC, s.location ASC) AS hottest_rank,
        RANK() OVER (ORDER BY s.avg_temp ASC, s.location ASC) AS coldest_rank
    FROM loc_summary s
    JOIN extremes_dates d ON s.location = d.location
    ORDER BY hottest_rank, s.location ASC

WITH base AS (
        SELECT *
        FROM weather_clean
        WHERE 1 = 1
    ),
    all_locations AS (
        SELECT DISTINCT location FROM base
    ),
    yearly_agg AS (
        SELECT
            location,
            year,
            AVG(temperature) AS avg_temp,
            CASE
                WHEN (CAST(COUNT(temperature) AS DOUBLE) / DATEDIFF(make_date(year + 1, 1, 1), make_date(year, 1, 1))) >= 0.95
                THEN true
                ELSE false
            END AS is_complete_year
        FROM base
        GROUP BY location, year
    ),
    complete_years AS (
        SELECT location, year, avg_temp
        FROM yearly_agg
        WHERE is_complete_year = true
    ),
    loc_regr AS (
        SELECT
            location,
            regr_slope(avg_temp, CAST(year AS DOUBLE)) AS raw_slope,
            regr_r2(avg_temp, CAST(year AS DOUBLE)) AS raw_r2,
            COUNT(*) AS n_years
        FROM complete_years
        GROUP BY location
    ),
    loc_endpoints AS (
        SELECT
            location,
            MAX(CASE WHEN rn_asc = 1 THEN avg_temp END) AS first_year_avg,
            MAX(CASE WHEN rn_desc = 1 THEN avg_temp END) AS last_year_avg
        FROM (
            SELECT
                location,
                avg_temp,
                ROW_NUMBER() OVER (PARTITION BY location ORDER BY year ASC) AS rn_asc,
                ROW_NUMBER() OVER (PARTITION BY location ORDER BY year DESC) AS rn_desc
            FROM complete_years
        )
        GROUP BY location
    ),
    loc_distinct AS (
        SELECT
            r.location,
            ROUND(r.raw_slope, 4) AS slope_c_per_year,
            ROUND(r.raw_slope * 10, 4) AS slope_per_decade,
            ROUND(COALESCE(r.raw_r2, 0.0), 4) AS r_squared,
            r.n_years,
            ROUND(e.first_year_avg, 2) AS first_year_avg,
            ROUND(e.last_year_avg, 2) AS last_year_avg
        FROM loc_regr r
        JOIN loc_endpoints e ON r.location = e.location
    ),
    loc_distinct_all AS (
        SELECT
            a.location,
            COALESCE(d.slope_c_per_year, 0.0) AS slope_c_per_year,
            COALESCE(d.slope_per_decade, 0.0) AS slope_per_decade,
            COALESCE(d.r_squared, 0.0) AS r_squared,
            COALESCE(d.n_years, 0) AS n_years,
            d.first_year_avg,
            d.last_year_avg
        FROM all_locations a
        LEFT JOIN loc_distinct d ON a.location = d.location
    ),
    all_loc_yearly AS (
        SELECT
            year,
            AVG(avg_temp) AS avg_temp
        FROM complete_years
        GROUP BY year
    ),
    all_loc_regr AS (
        SELECT
            'All locations' AS location,
            regr_slope(avg_temp, CAST(year AS DOUBLE)) AS raw_slope,
            regr_r2(avg_temp, CAST(year AS DOUBLE)) AS raw_r2,
            COUNT(*) AS n_years
        FROM all_loc_yearly
    ),
    all_loc_endpoints AS (
        SELECT
            'All locations' AS location,
            MAX(CASE WHEN rn_asc = 1 THEN avg_temp END) AS first_year_avg,
            MAX(CASE WHEN rn_desc = 1 THEN avg_temp END) AS last_year_avg
        FROM (
            SELECT
                avg_temp,
                ROW_NUMBER() OVER (ORDER BY year ASC) AS rn_asc,
                ROW_NUMBER() OVER (ORDER BY year DESC) AS rn_desc
            FROM all_loc_yearly
        )
    ),
    all_loc_distinct AS (
        SELECT
            r.location,
            ROUND(COALESCE(r.raw_slope, 0.0), 4) AS slope_c_per_year,
            ROUND(COALESCE(r.raw_slope, 0.0) * 10, 4) AS slope_per_decade,
            ROUND(COALESCE(r.raw_r2, 0.0), 4) AS r_squared,
            r.n_years,
            ROUND(e.first_year_avg, 2) AS first_year_avg,
            ROUND(e.last_year_avg, 2) AS last_year_avg
        FROM all_loc_regr r
        JOIN all_loc_endpoints e ON r.location = e.location
    ),
    combined AS (
        SELECT * FROM loc_distinct_all
        UNION ALL
        SELECT * FROM all_loc_distinct
    )
    SELECT
        location,
        slope_c_per_year,
        slope_per_decade,
        r_squared,
        n_years,
        first_year_avg,
        last_year_avg,
        CASE
            WHEN n_years < 5 THEN 'insufficient data'
            WHEN r_squared < 0.10 THEN 'no clear trend'
            WHEN slope_c_per_year > 0 THEN 'warming'
            ELSE 'cooling'
        END AS trend_label
    FROM combined
    ORDER BY CASE WHEN location = 'All locations' THEN 1 ELSE 0 END, location ASC

WITH full_thresholds AS (
        SELECT
            location,
            percentile_approx(temp_max, 0.95, 10000) AS hot_thresh
        FROM weather_clean
        GROUP BY location
    ),
    filtered_base AS (
        SELECT *
        FROM weather_clean
        WHERE 1 = 1
    ),
    flagged_days AS (
        SELECT
            b.location,
            b.date,
            b.temp_max,
            DATE_SUB(b.date, CAST(ROW_NUMBER() OVER (PARTITION BY b.location ORDER BY b.date) AS INT)) AS spell_grp
        FROM filtered_base b
        JOIN full_thresholds t ON b.location = t.location
        WHERE b.temp_max >= t.hot_thresh
    ),
    spells AS (
        SELECT
            location,
            MIN(date) AS spell_start,
            MAX(date) AS spell_end,
            COUNT(*) AS length_days,
            ROUND(MAX(temp_max), 2) AS peak_temp_max
        FROM flagged_days
        GROUP BY location, spell_grp
        HAVING COUNT(*) >= 3
    )
    SELECT
        location,
        CAST(spell_start AS STRING) AS spell_start,
        CAST(spell_end AS STRING) AS spell_end,
        length_days,
        peak_temp_max
    FROM spells
    ORDER BY length_days DESC, peak_temp_max DESC, location ASC

WITH base AS (
        SELECT *
        FROM weather_clean
        WHERE 1 = 1
    ),
    headline_aggs AS (
        SELECT
            COUNT(*) AS total_records,
            COUNT(DISTINCT location) AS locations,
            CAST(MIN(date) AS STRING) AS first_date,
            CAST(MAX(date) AS STRING) AS last_date,
            MIN(date) AS raw_first_date,
            MAX(date) AS raw_last_date,
            ROUND(AVG(temperature), 2) AS avg_temp,
            MAX(temp_max) AS max_temp_raw,
            MIN(temp_min) AS min_temp_raw,
            ROUND(COALESCE(SUM(rainfall), 0.0), 2) AS total_rainfall_mm,
            ROUND(AVG(humidity), 2) AS avg_humidity,
            ROUND(AVG(wind_speed), 2) AS avg_wind_speed,
            ROUND(AVG(pressure), 2) AS avg_pressure,
            COALESCE(SUM(CASE WHEN `temperature_was_filled` THEN 1 ELSE 0 END), 0) + COALESCE(SUM(CASE WHEN `temp_max_was_filled` THEN 1 ELSE 0 END), 0) + COALESCE(SUM(CASE WHEN `temp_min_was_filled` THEN 1 ELSE 0 END), 0) + COALESCE(SUM(CASE WHEN `humidity_was_filled` THEN 1 ELSE 0 END), 0) + COALESCE(SUM(CASE WHEN `wind_speed_was_filled` THEN 1 ELSE 0 END), 0) + COALESCE(SUM(CASE WHEN `pressure_was_filled` THEN 1 ELSE 0 END), 0) AS interpolated_value_count
        FROM base
    ),
    hottest_day AS (
        SELECT
            CAST(date AS STRING) AS max_temp_date,
            location AS max_temp_location
        FROM base
        WHERE temp_max = (SELECT max_temp_raw FROM headline_aggs)
        ORDER BY date ASC, location ASC
        LIMIT 1
    ),
    coldest_day AS (
        SELECT
            CAST(date AS STRING) AS min_temp_date,
            location AS min_temp_location
        FROM base
        WHERE temp_min = (SELECT min_temp_raw FROM headline_aggs)
        ORDER BY date ASC, location ASC
        LIMIT 1
    ),
    yearly_totals AS (
        SELECT
            year,
            AVG(temperature) AS y_temp,
            SUM(rainfall) AS y_rain
        FROM base
        GROUP BY year
    ),
    hottest_y AS (
        SELECT year AS hottest_year
        FROM yearly_totals
        ORDER BY y_temp DESC, year ASC
        LIMIT 1
    ),
    wettest_y AS (
        SELECT year AS wettest_year
        FROM yearly_totals
        ORDER BY y_rain DESC, year ASC
        LIMIT 1
    )
    SELECT
        h.total_records,
        h.locations,
        h.first_date,
        h.last_date,
        h.avg_temp,
        ROUND(h.max_temp_raw, 2) AS max_temp,
        t_hot.max_temp_date,
        t_hot.max_temp_location,
        ROUND(h.min_temp_raw, 2) AS min_temp,
        t_cold.min_temp_date,
        t_cold.min_temp_location,
        h.total_rainfall_mm,
        ROUND(
            h.total_rainfall_mm / (
                NULLIF(h.locations, 0) * (
                    GREATEST(DATEDIFF(h.raw_last_date, h.raw_first_date) + 1, 1) / 365.25
                )
            ),
            2
        ) AS avg_annual_rainfall_per_location,
        h.avg_humidity,
        h.avg_wind_speed,
        h.avg_pressure,
        wy.wettest_year,
        hy.hottest_year,
        h.interpolated_value_count
    FROM headline_aggs h
    CROSS JOIN hottest_day t_hot
    CROSS JOIN coldest_day t_cold
    CROSS JOIN hottest_y hy
    CROSS JOIN wettest_y wy
